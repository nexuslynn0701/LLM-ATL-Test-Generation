-- [articleSet contains a subset of Article, so that each journal

--   * a section_1 paragraph;

				if ret->collect(e | e.journal)->includes(e.journal) then
					ret
				else
					ret->including(e)
				endif

		)

		entry_para : DocBook!Para (
			content <- '[' + e.id + ']'
