OCL CONSTRAINTS
--------------------------------
All generated source models must satisfy the following constraints:
-- Grafcet2PetriNet OCL Contract
-- Transformation-level contract: source-side (Grafcet) to target-side (PetriNet) mapping.
-- Written in the style of Troya et al. 2015 TSE.

-- C1 (Rule PetriNet): For each Grafcet, a PetriNet must be created with the same name and location.
context Grafcet!Grafcet
inv PetriNetCreated:
  PetriNet!PetriNet.allInstances()->exists(p |
    p.name = self.name and p.location = self.location
  )

-- C2 (Rule PetriNet): The number of PetriNet instances must equal the number of Grafcet instances.
context Grafcet!Grafcet
inv PetriNetCount:
  PetriNet!PetriNet.allInstances()->size() = Grafcet!Grafcet.allInstances()->size()

-- C3 (Rule Place): For each Step, a Place must be created with the same name and location.
context Grafcet!Step
inv PlaceCreated:
  PetriNet!Place.allInstances()->exists(p |
    p.name = self.name and p.location = self.location
  )

-- C4 (Rule Place): The number of Place instances must equal the number of Step instances.
context Grafcet!Step
inv PlaceCount:
  PetriNet!Place.allInstances()->size() = Grafcet!Step.allInstances()->size()

-- C5 (Rule Place): For each Step marked as isInitial=true, the corresponding Place must have initToken=true.
context Grafcet!Step
inv InitialStepBecomesInitialPlace:
  self.isInitial implies
  PetriNet!Place.allInstances()->exists(p |
    p.name = self.name and p.initToken = true
  )

-- C6 (Rule Transition): For each Grafcet Transition, a PetriNet Transition must be created with the same name.
context Grafcet!Transition
inv TransitionCreated:
  PetriNet!Transition.allInstances()->exists(t |
    t.name = self.name
  )

-- C7 (Rule Transition): The number of PetriNet Transition instances must equal the number of Grafcet Transition instances.
context Grafcet!Transition
inv TransitionCount:
  PetriNet!Transition.allInstances()->size() = Grafcet!Transition.allInstances()->size()

-- C8 (Rule PlaceToTransition): For each StepToTransition, a PlaceToTransition Arc must connect the corresponding Place and Transition.
context Grafcet!StepToTransition
inv ArcCreated:
  PetriNet!PlaceToTransition.allInstances()->exists(a |
    a.from.name = self.from.name and
    a.to.name = self.to.name and
    a.name = self.name
  )

-- C9 (Rule TransitionToPlace): For each TransitionToStep, a TransitionToPlace Arc must connect the corresponding Transition and Place.
context Grafcet!TransitionToStep
inv ArcCreated:
  PetriNet!TransitionToPlace.allInstances()->exists(a |
    a.from.name = self.from.name and
    a.to.name = self.to.name and
    a.name = self.name
  )

-- C10 (Rule PlaceToTransition + Rule TransitionToPlace): The total number of Arcs must equal the total number of Connections.
context PetriNet!PetriNet
inv ArcConnectionCount:
  self.arcs->size() = Grafcet!Grafcet.allInstances()->collect(g | g.connections)->flatten()->size()

module Grafcet2PetriNet;
create OUT : PetriNet, TM : TMM from IN : Grafcet;

helper def : trace : TMM!Trace = OclUndefined;

-------------------------------------------------------------------------------
-- RULES ----------------------------------------------------------------------
-------------------------------------------------------------------------------
entrypoint rule createTrace()
{
	to
		tr : TMM!Trace( )

	do {
		thisModule.trace <- tr;
	}
}

-- The PetriNet rule generates a PetriNet element from the input Grafcet element. 
-- Name of the generated PetriNet element is copied from the one of the input Grafcet.
-- Its set of Elements corresponds to Elements generated by Place and Transition rules.
-- And its set of Arcs corresponds to Connections generated by PlaceToTransition and TransitionToPlace rules.
rule PetriNet {
	from
		g : Grafcet!Grafcet
	to
		p : PetriNet!PetriNet
		(
			location <- g.location,
			name <- g.name,
			elements <- g.elements,
			arcs <- g.connections
		),
		-- trace information
		tl : TMM!TraceLink(
			ruleName <- 'PetriNet',
			targetElements <- Sequence{p}
		)
	do {
		-- trace information
		tl.sourceElements <- Sequence{g};
		thisModule.trace.traceLinks <- tl;
	}
}

--------------------------------
SOURCE FORMAT NOTES
--------------------------------
Generate only Grafcet source models.

Use an xmi:XMI root containing exactly one Grafcet:Grafcet element.

Declare:
- xmlns:xmi="http://www.omg.org/XMI"
- xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
- xmlns:Grafcet="Grafcet"

Represent contained nodes under Grafcet:Grafcet as:
- <elements xsi:type="Grafcet:Step" .../>
- <elements xsi:type="Grafcet:Transition" .../>
- <connections xsi:type="Grafcet:StepToTransition" .../>
- <connections xsi:type="Grafcet:TransitionToStep" .../>

Use positional references for connection endpoints:
- from="/0/@elements.N"
- to="/0/@elements.N"

Every StepToTransition must point from a Step to a Transition.
Every TransitionToStep must point from a Transition to a Step.

Do not explicitly serialize inverse references:
- grafcet
- incomingConnections
- outgoingConnections

Do not instantiate PetriNet target classes.
Do not generate target models.
