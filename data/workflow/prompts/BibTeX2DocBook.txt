Study this ATL transformation carefully to understand which source structures
trigger which rules. Use this knowledge to design a source model that exercises

module BibTeX2DocBook;
create OUT : DocBook, TM : TMM from IN : BibTeX;

helper def : trace : TMM!Trace = OclUndefined;

-------------------------------------------------------------------------------
-- RULES ----------------------------------------------------------------------
-------------------------------------------------------------------------------
entrypoint rule createTrace()
{
	to
		tr : TMM!Trace( )
	
	do {
		thisModule.trace <- tr;
	}	
}

-- R1
-- Rule 'Main'
-- This rule generates the structure of the DocBook model from a BibTeXFile element
rule Main {
	from
        bib : BibTeX!BibTeXFile
	to 
		doc : DocBook!DocBook (
			books <- boo
		),
		boo : DocBook!Book (
			articles <- art
		),
		art : DocBook!Article (
			title <- 'BibTeXML to DocBook',
			sections_1 <- Sequence{se1, se2, se3, se4}
		),
		se1 : DocBook!Sect1 (
			title <- 'References List',
			paras <- BibTeX!BibTeXEntry.allInstances()->sortedBy(e | e.id)
		),
		se2 : DocBook!Sect1 (
			title <- 'Authors List',
			paras <- BibTeX!Author.allInstances()->iterate(e; ret : Sequence(BibTeX!Author) = Sequence {} |
				if ret->collect(e | e.author)->includes(e.author) then
					ret
				else
					ret->including(e)
				endif
			)->sortedBy(e | e.author)
		),
		se3 : DocBook!Sect1 (
			title <- 'Titles List',
			paras <- BibTeX!TitledEntry.allInstances()->iterate(e; ret : Sequence(BibTeX!TitledEntry) = Sequence {} |
				if ret->collect(e | e.title)->includes(e.title) then
					ret
				else
					ret->including(e)
				endif
			)->sortedBy(e | e.title)	->collect(e | thisModule.resolveTemp(e, 'title_para'))
		),
		se4 : DocBook!Sect1 (
			title <- 'Journals List',
			paras <- BibTeX!Article.allInstances()->iterate(e; ret : Sequence(BibTeX!Article) = Sequence {} |
				if ret->collect(e | e.journal)->includes(e.journal) then
					ret
				else
					ret->including(e)
				endif
			)->sortedBy(e | e.journal)->collect(e | thisModule.resolveTemp(e, 'journal_para'))
		),
		-- trace information
		tl : TMM!TraceLink(
			ruleName <- 'Main',
			targetElements <- Sequence{doc,boo,art,se1,se2,se3,se4}
		)
	do {
		-- trace information
		tl.sourceElements <- Sequence{bib};
		thisModule.trace.traceLinks <- tl;
	}
}

-- R2
-- Rule 'Author'
-- This rule generates a section_2 paragraph for each distinct author.
rule Author {
	from
		a : BibTeX!Author (
			  BibTeX!Author.allInstances()->iterate(e; ret : Sequence(BibTeX!Author) = Sequence {} |
				if ret->collect(e | e.author)->includes(e.author) then
					ret
				else
					ret->including(e)
				endif
			)->sortedBy(e | e.author)->includes(a))
	to
		p1 : DocBook!Para (
			content <- a.author
		),
		-- trace information
		tl : TMM!TraceLink(
			ruleName <- 'Author',
			targetElements <- Sequence{p1}
		)
	do {
		-- trace information
		tl.sourceElements <- Sequence{a};
		thisModule.trace.traceLinks <- tl;
	}
}

-- R3
-- Rule 'UntitledEntry'
-- This rule generates a section_1 paragraph for each untitled entry.
-- The elements that should trigger this rule are: Misc, Booklet
rule UntitledEntry {
	from
		e : BibTeX!BibTeXEntry (
			not e.oclIsKindOf(BibTeX!TitledEntry)
		)
	to
		p : DocBook!Para (
			content <- '[' + e.id + ']'
				+ ' ' + e.oclType().name
				+ (if e.oclIsKindOf(BibTeX!TitledEntry) then ' ' + e.title else '' endif)
				+ (if e.oclIsKindOf(BibTeX!AuthoredEntry) 
				   then e.authors->iterate(e; str : String = '' | str + ' ' + e.author)

--------------------------------
GENERATION REQUIREMENTS
--------------------------------
The source model must:
1. Conform to the source metamodel.
2. Satisfy all OCL constraints.
3. Exercise as many different ATL rules as possible.
4. Be structurally different from the example source models.
5. Be syntactically valid XMI.
6. Use ONLY classes, attributes, references, and namespace prefixes defined
   in the source metamodel.
7. Never instantiate target metamodel classes.
8. Express attribute values as XMI attributes, not as XML text content nodes.
   CORRECT:   <Entry title="My Title" year="2024"/>
   INCORRECT: <Entry><title>My Title</title></Entry>
9. Containment references may be represented as nested XMI elements when
   required by the source metamodel.

--------------------------------
SOURCE FORMAT NOTES
--------------------------------

CRITICAL XMI FORMAT for BibTeX2DocBook source models:

1. Root element MUST be <BibTeX:BibTeXFile> directly — NOT wrapped in <xmi:XMI>:
   CORRECT:
     <BibTeX:BibTeXFile xmi:version="2.0"
         xmlns:xmi="http://www.omg.org/XMI"
         xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
         xmlns:BibTeX="http://BibTeX/1.0">
   INCORRECT:
     <xmi:XMI xmi:version="2.0" ...>
       <BibTeX:BibTeXFile>

2. The xmlns:xsi namespace MUST be declared on the root element.
   xsi:type is used on <entries> elements and requires this declaration.

3. All <entries> elements use xsi:type to specify the concrete type:
   <entries xsi:type="BibTeX:Article" id="..." title="..." year="..." journal="...">

4. Authors are nested elements, not attributes:
   <authors author="John Doe"/>

OUTPUT FORMAT
--------------------------------
Return exactly one source model and nothing else:

<source_model>
[source XMI here]
</source_model>

Do not include: target_model, expected output, coverage notes,
explanation, or Markdown fences.
